Membuka modal dan mengisi form sesuai barang yang dipilih function bukaModalEdit(id) { // Cari item berdasarkan id di array database kamu (sesuaikan nama variabel array) const barang = databaseBarang.find(item => item.id === id); if (barang) { document.getElementById('editBarangId').value = barang.id; document.getElementById('editNamaBarang').value = barang.nama; document.getElementById('editHargaBeli').value = barang.hargaBeli; document.getElementById('editHargaJual').value = barang.hargaJual; document.getElementById('modalEditBarang').style.display = 'flex'; } } // 2. Menutup modal function tutupModalEdit() { document.getElementById('modalEditBarang').style.display = 'none'; } // 3. Menyimpan perubahan ke database/array & update tampilan function simpanPerubahanBarang() { const id = document.getElementById('editBarangId').value; const hargaBeliBaru = parseFloat(document.getElementById('editHargaBeli').value) || 0; const hargaJualBaru = parseFloat(document.getElementById('editHargaJual').value) || 0; const index = databaseBarang.findIndex(item => item.id === id); if (index !== -1) { // Update data di memori databaseBarang[index].hargaBeli = hargaBeliBaru; databaseBarang[index].hargaJual = hargaJualBaru; // Simpan ke localStorage (sesuaikan key localStorage kamu) localStorage.setItem('databaseBarang', JSON.stringify(databaseBarang)); // Render ulang tabel database agar harga langsung terbarui renderTabelDatabase(); tutupModalEdit(); alert('Harga beli dan harga jual berhasil diperbarui!'); } } // Contoh potongan pembuatan baris tabel di JavaScript: let html = ''; databaseBarang.forEach(barang => { html += ` ${barang.nama} Rp ${barang.hargaBeli.toLocaleString('id-ID')} Rp ${barang.hargaJual.toLocaleString('id-ID')} `; }); document.getElementById('tabelDatabaseBody').innerHTML = html;
